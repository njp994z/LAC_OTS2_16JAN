import math
from datetime import datetime

CONFIG = {
    "pipe_dia_in": 4.0,
    "line_length_ft": 80.0,
    "friction_factor": 0.018,
    "K_minor_losses": 7.5,
    "SG": 1.79,
    "barometric_psia": 14.696,
    "temperature_f": 275.0,

    # Pump curve (static)
    "pump_coeffs": [-1.06594794e-05, -7.73601399e-04, -4.91452991e-02, 296.104895],

    # Valve equal-%
    "Cv_max": 548.0,
    "Cv_at_pos": 25.039,
    "pos_at_Cv_pct": 30.6,

    # Plant boundary
    "furnace_static_psi": 5.0,
    "discharge_line_loss_psi": 2.0,

    # Nozzle modeled as Cv element (keeps ΔP ~ Q^2)
    "Cv_nozzle": 24.98,

    # Datasheet calibration target for valve ΔP at max case
    "cal_flow_gpm": 86.91,            # 19.74 m3/h
    "cal_dp_valve_psi": 21.66,        # 149.34 kPa
}

class StaticSulfurSprayHydraulics:
    def __init__(self):
        self.g = 32.174
        self.pipe_dia_ft = CONFIG["pipe_dia_in"] / 12.0
        self.pipe_area_ft2 = math.pi * (self.pipe_dia_ft / 2.0) ** 2
        self.psi_per_ft = CONFIG["SG"] * 0.433
        self.pump_coeffs = CONFIG["pump_coeffs"]

        self.R = float(CONFIG.get("R_equal_percent", self._compute_R_from_point()))
        self.R = max(self.R, 1.000001)

        # ✅ Calibrate missing downstream header pressure using the datasheet valve ΔP target
        self.downstream_header_psig = self._calibrate_downstream_header_psig()

    def _compute_R_from_point(self) -> float:
        Cv_max = max(float(CONFIG["Cv_max"]), 1e-12)
        Cv1 = max(float(CONFIG["Cv_at_pos"]), 1e-12)
        p1 = float(CONFIG["pos_at_Cv_pct"]) / 100.0
        return (Cv1 / Cv_max) ** (1.0 / (p1 - 1.0))

    @staticmethod
    def _clamp(x: float, lo: float, hi: float) -> float:
        return max(lo, min(hi, x))

    def nozzle_dp_psi(self, Q_gpm: float) -> float:
        Q = max(0.0, float(Q_gpm))
        if Q <= 0.0:
            return 0.0
        Cv_n = max(float(CONFIG["Cv_nozzle"]), 1e-12)
        SG = float(CONFIG["SG"])
        return SG * (Q / Cv_n) ** 2

    def pos_from_Cv(self, Cv: float) -> float:
        Cv = max(float(Cv), 0.0)
        Cv_max = max(float(CONFIG["Cv_max"]), 1e-12)
        Cv_min = Cv_max / self.R
        if Cv <= Cv_min:
            return 0.0
        if Cv >= Cv_max:
            return 100.0
        return 100.0 * (1.0 + math.log(Cv / Cv_max) / math.log(self.R))

    def _valve_inlet_psig_from_flow(self, Q: float, pit_level_ft: float) -> float:
        a, b, c, d = self.pump_coeffs
        pump_head_ft = max(0.0, a * Q**3 + b * Q**2 + c * Q + d)

        if Q > 0.0:
            Q_cfs = Q / 448.831
            v_fps = Q_cfs / self.pipe_area_ft2
            h_f_ft = (
                (CONFIG["friction_factor"] * CONFIG["line_length_ft"] / self.pipe_dia_ft + CONFIG["K_minor_losses"])
                * (v_fps**2)
                / (2.0 * self.g)
            )
        else:
            h_f_ft = 0.0

        suction_psig = pit_level_ft * self.psi_per_ft
        pump_discharge_psig = suction_psig + (pump_head_ft - h_f_ft) * self.psi_per_ft
        return pump_discharge_psig - float(CONFIG["discharge_line_loss_psi"])

    def _calibrate_downstream_header_psig(self, pit_level_ft: float = 7.0) -> float:
        Qc = float(CONFIG["cal_flow_gpm"])
        dp_valve_target = float(CONFIG["cal_dp_valve_psi"])

        P1 = self._valve_inlet_psig_from_flow(Qc, pit_level_ft)
        dp_noz = self.nozzle_dp_psi(Qc)

        return P1 - float(CONFIG["furnace_static_psi"]) - dp_noz - dp_valve_target

    def calculate(self, flow_gpm: float, pit_level_ft: float = 7.0) -> dict:
        Q = max(0.0, float(flow_gpm))

        P1 = self._valve_inlet_psig_from_flow(Q, pit_level_ft)
        dp_noz = self.nozzle_dp_psi(Q)
        P2 = self.downstream_header_psig + float(CONFIG["furnace_static_psi"]) + dp_noz

        dp_valve = max(P1 - P2, 0.0)
        Cv_req = Q * math.sqrt(float(CONFIG["SG"]) / dp_valve) if (Q > 0 and dp_valve > 0) else 0.0

        pos_pct = self._clamp(self.pos_from_Cv(Cv_req), 0.0, 100.0)
        pid_mA = 4.0 + 16.0 * (pos_pct / 100.0)

        return {
            "Flow_gpm": round(Q, 2),
            "Cv_required": round(Cv_req, 6),
            "valve_position_pct": round(pos_pct, 4),
            "pid_output_mA": round(pid_mA, 4),
            "valve_inlet_psig": round(P1, 2),
            "valve_outlet_psig": round(P2, 2),
            "dP_valve_psi": round(dp_valve, 3),
            "downstream_header_psig": round(self.downstream_header_psig, 3),
            "timestamp": datetime.now().isoformat(),
        }
