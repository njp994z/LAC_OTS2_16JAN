import math
from datetime import datetime
import numpy as np
from scipy.interpolate import interp1d  # For smooth interpolation

# Shared configuration (no R needed anymore)
CONFIG = {
    'pipe_dia_in': 4.0,
    'line_length_ft': 80.0,
    'deltaP_nozzle_psi': 150.0,
    'furnace_static_psi': 7.0,
    'friction_factor': 0.018,
    'K_minor_losses': 7.5,
    'SG': 1.79,
    'Cv_max': 548.0,
    'barometric_psig': 0.0,
    'temperature_f': 275.0
}

class StaticSulfurSprayHydraulics:
    """Static calculation to populate DCS material balance table"""

    def __init__(self):
        self.g = 32.174
        self.psi_per_ft = CONFIG['SG'] * 0.433
        self.pipe_dia_ft = CONFIG['pipe_dia_in'] / 12.0
        self.pipe_area_ft2 = math.pi * (self.pipe_dia_ft / 2) ** 2

        # Cubic fit from Weir/Lewis curve (gives ~285 ft at 81 gpm)
        self.pump_coeffs = [-1.06594794e-05, -7.73601399e-04, -4.91452991e-02, 296.104895]

        # Datasheet points for equal % valve: Cv vs % open
        self.cv_points = np.array([4.549, 12.169, 25.039])
        self.pos_points = np.array([16.27, 23.4, 30.6])

        # Interpolation function: Cv → % open (linear for simplicity, can use cubic if desired)
        self.cv_to_pos = interp1d(self.cv_points, self.pos_points, kind='linear',
                                  fill_value="extrapolate", bounds_error=False)

    def calculate(self, flow_gpm: float, pit_level_ft: float = 7.0) -> dict:
        Q = max(0.1, flow_gpm)

        # Pump head
        a, b, c, d = self.pump_coeffs
        pump_head_ft = a * Q**3 + b * Q**2 + c * Q + d

        # Friction loss
        Q_cfs = Q / 448.831
        v_fps = Q_cfs / self.pipe_area_ft2
        h_f = (CONFIG['friction_factor'] * CONFIG['line_length_ft'] / self.pipe_dia_ft +
               CONFIG['K_minor_losses']) * (v_fps**2) / (2 * self.g)

        # Suction pressure
        suction_psig = pit_level_ft * self.psi_per_ft + CONFIG['barometric_psig']

        # Pump outlet pressure
        pump_discharge_psig = suction_psig + pump_head_ft * self.psi_per_ft - h_f * self.psi_per_ft

        # Nozzle inlet pressure = furnace + nozzle ΔP
        nozzle_inlet_psig = CONFIG['furnace_static_psi'] + CONFIG['deltaP_nozzle_psi']  # 157 psig

        # Discharge line loss (pump outlet to valve inlet)
        discharge_line_loss_psi = 2.0
        valve_inlet_psig = pump_discharge_psig - discharge_line_loss_psi

        # Valve ΔP
        dp_valve_psi = max(valve_inlet_psig - nozzle_inlet_psig, 0.01)

        # Required Cv
        Cv_req = Q / math.sqrt(dp_valve_psi / CONFIG['SG'])

        # Valve position % - interpolate from datasheet points
        if Cv_req <= self.cv_points[0]:
            pos_pct = self.pos_points[0]  # Extrapolate low
        elif Cv_req >= self.cv_points[-1]:
            pos_pct = self.pos_points[-1]  # Extrapolate high (conservative)
        else:
            pos_pct = self.cv_to_pos(Cv_req)

        # Back-calculate PID controller output (4-20 mA → 0-100% positioner)
        pid_output_mA = 4.0 + (pos_pct / 100.0) * 16.0
        pid_output_mA = max(4.0, min(20.0, pid_output_mA))

        # Convert pressures to psia
        pump_discharge_psia = pump_discharge_psig + 14.7
        valve_inlet_psia = valve_inlet_psig + 14.7
        nozzle_inlet_psia = nozzle_inlet_psig + 14.7

        # Mass flow
        mass_lb_per_hr = Q * CONFIG['SG'] * 500.3
        mass_klb_per_hr = mass_lb_per_hr / 1000

        return {
            "1540-PI-2600_psia": round(pump_discharge_psia, 2),
            "1540-FIC-2602_gpm": round(Q, 2),
            "1540-PI-2604_psia": round(nozzle_inlet_psia, 2),

            "x_H2SO4": 0.00,
            "x_H2O": 0.00,
            "x_Sulfur": 1.00,
            "m_Total_klb_hr": round(mass_klb_per_hr, 2),

            "Pressure_psia_pump": round(pump_discharge_psia, 2),
            "Pressure_psia_valve": round(valve_inlet_psia, 2),
            "Pressure_psia_nozzle": round(nozzle_inlet_psia, 2),

            "Temp_F": CONFIG['temperature_f'],
            "Flow_gpm_pump": round(Q, 2),
            "Flow_gpm_valve": round(Q, 2),
            "Flow_gpm_nozzle": round(Q, 2),

            # Additional diagnostics
            "pump_head_ft": round(pump_head_ft, 2),
            "valve_position_pct": round(pos_pct, 2),
            "Cv_used": round(Cv_req, 2),
            "dp_valve_psi": round(dp_valve_psi, 2),
            "pid_output_mA": round(pid_output_mA, 2),  # Back-calculated PID output
            "timestamp": datetime.now().isoformat()
        }
