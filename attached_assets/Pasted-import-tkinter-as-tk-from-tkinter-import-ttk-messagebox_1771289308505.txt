# Double-click to edit any cell (except Parameter column)
edit_entry = None
def on_double_click(event):
    global edit_entry
    if edit_entry: edit_entry.destroy()

    item = inlet_tree.identify_row(event.y)
    col = inlet_tree.identify_column(event.x)
    if not item or col == "#1": return  # skip Parameter

    x, y, w, h = inlet_tree.bbox(item, col)
    edit_entry = Entry(inlet_frame, bg="#3a3a3a", fg="#ffffff", font=("Segoe UI", 10))
    edit_entry.place(x=x, y=y, width=w, height=h)
    edit_entry.insert(0, inlet_tree.item(item)["values"][int(col[1:]) - 1])
    edit_entry.focus()

    def save_edit(e=None):
        global edit_entry
        if edit_entry:
            new_val = edit_entry.get()
            values = list(inlet_tree.item(item)["values"])
            values[int(col[1:]) - 1] = new_val
            inlet_tree.item(item, values=values)
            edit_entry.destroy()
            edit_entry = None

    edit_entry.bind("<Return>", save_edit)
    edit_entry.bind("<FocusOut>", save_edit)

inlet_tree.bind("<Double-1>", on_double_click)

# ── Calculate function ───────────────────────────────────────────
def run_static_calculation():
    try:
        # Get inlet values
        inlet_values = {inlet_tree.item(item)["values"][0]: inlet_tree.item(item)["values"] for item in inlet_tree.get_children()}

        t12 = float(inlet_values["TEMPERATURE"][2])
        t14 = float(inlet_values["TEMPERATURE"][3])
        t17a = float(inlet_values["TEMPERATURE"][4])
        total12 = float(inlet_values["TOTAL"][2].replace(",", ""))
        total14 = float(inlet_values["TOTAL"][3].replace(",", ""))
        total17a = float(inlet_values["TOTAL"][4].replace(",", ""))

        target3 = float(pass3_target.get())
        target4 = float(pass4_target.get())

        # Placeholder calculation (replace with real backend)
        hip_out = target3 - 10
        cip_out = target4 - 10
        hip_bypass_frac = 0.35
        cip_bypass_frac = 0.42
        hip_bypass = total12 * hip_bypass_frac
        cip_bypass = total14 * cip_bypass_frac
        hip_hx = total12 - hip_bypass
        cip_hx = total14 - cip_bypass

        # Rough duty (Q = m * Cp * ΔT, Cp≈0.25, MW≈34, 60 min/hr → MMBTU/hr)
        duty_hip = (hip_hx * 60 * 34 / 379 * 0.25 * 80) / 1_000_000
        duty_cip = (cip_hx * 60 * 34 / 379 * 0.25 * 70) / 1_000_000

        hip_pos = 65.0
        cip_36_pos = 100 if hip_pos > 50 else 0
        cip_78_pos = max(0, (hip_pos - 50) * 2)

        # Update HIP table
        hip_tree.delete(*hip_tree.get_children())
        hip_tree.insert("", "end", values=("TEMPERATURE", "°F", f"{t12:.1f}", f"{hip_out:.1f}", f"{t17a:.1f}", f"{t17a:.1f}", f"{hip_bypass:,.0f}", f"{target3:.1f}", f"{t17a + 120:.1f}", f"{target3:.1f}"))

        # Update CIP table
        cip_tree.delete(*cip_tree.get_children())
        cip_tree.insert("", "end", values=("TEMPERATURE", "°F", f"{t14:.1f}", f"{cip_out:.1f}", f"{t17a:.1f}", f"{t17a:.1f}", f"{cip_bypass:,.0f}", f"{cip_bypass:,.0f}", f"{t17a + 140:.1f}", f"{target4:.1f}"))

        # Update extra rows
        extra_labels = ["HIP 48\" Valve Position", "CIP 36\" Valve Position", "CIP 78\" Valve Position", "HIP HX / Bypass ratio", "CIP HX / Bypass ratio", "HIP HX Duty", "CIP HX Duty"]
        extra_values = [f"{hip_pos:.1f}", f"{cip_36_pos:.1f}", f"{cip_78_pos:.1f}", f"{hip_hx/total12:.3f}", f"{cip_hx/total14:.3f}", f"{duty_hip:.2f}", f"{duty_cip:.2f}"]
        for i, (label, val) in enumerate(zip(extra_labels, extra_values)):
            row = extra_frame.winfo_children()[i]
            row.winfo_children()[2].config(text=val)

        messagebox.showinfo("Calculation Complete", "Output streams and HX duties updated.")

    except Exception as e:
        messagebox.showerror("Error", str(e))
