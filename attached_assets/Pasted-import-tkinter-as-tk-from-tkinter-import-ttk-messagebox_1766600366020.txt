import tkinter as tk
from tkinter import ttk, messagebox

from static_backend import SulfurSprayHydraulics as StaticModel
from dynamic_backend import DynamicSulfurSprayHydraulics as DynamicModel

root = tk.Tk()
root.title("Sulfur Hydraulics Simulator")
root.geometry("600x500")

notebook = ttk.Notebook(root)
notebook.pack(fill="both", expand=True)

# Static Tab
static_tab = ttk.Frame(notebook)
notebook.add(static_tab, text="Static")

tk.Label(static_tab, text="Flow (GPM):").grid(row=0, column=0, padx=10, pady=5, sticky="e")
flow_entry = tk.Entry(static_tab)
flow_entry.grid(row=0, column=1, padx=10, pady=5)
flow_entry.insert(0, "81.35")

tk.Label(static_tab, text="Pit Level (ft):").grid(row=1, column=0, padx=10, pady=5, sticky="e")
pit_entry = tk.Entry(static_tab)
pit_entry.grid(row=1, column=1, padx=10, pady=5)
pit_entry.insert(0, "7.0")

output_text = tk.Text(static_tab, height=20, width=60)
output_text.grid(row=3, column=0, columnspan=3, padx=10, pady=10)

def calc_static():
    try:
        model = StaticModel()
        flow = float(flow_entry.get())
        pit = float(pit_entry.get())
        res = model.calculate(flow, pit)
        output_text.delete(1.0, tk.END)
        for k, v in res.items():
            output_text.insert(tk.END, f"{k}: {v}\n")
        return res
    except ValueError:
        messagebox.showerror("Error", "Invalid input values")

def save_static():
    res = calc_static()
    if "status" not in res or res["status"] != "error":
        model = StaticModel()
        model.save_to_sqlite(res)
        messagebox.showinfo("Success", "Saved to database")

calc_button = tk.Button(static_tab, text="Calculate", command=calc_static)
calc_button.grid(row=2, column=0, pady=10)

save_button = tk.Button(static_tab, text="Save to DB", command=save_static)
save_button.grid(row=2, column=1, pady=10)

# Dynamic Tab
dynamic_tab = ttk.Frame(notebook)
notebook.add(dynamic_tab, text="Dynamic")

tk.Label(dynamic_tab, text="Setpoint (GPM):").grid(row=0, column=0, padx=10, pady=5, sticky="e")
setpoint_entry = tk.Entry(dynamic_tab)
setpoint_entry.grid(row=0, column=1, padx=10, pady=5)
setpoint_entry.insert(0, "81.35")

tk.Label(dynamic_tab, text="Pit Level (ft):").grid(row=1, column=0, padx=10, pady=5, sticky="e")
pit_d_entry = tk.Entry(dynamic_tab)
pit_d_entry.grid(row=1, column=1, padx=10, pady=5)
pit_d_entry.insert(0, "7.0")

time_label = tk.Label(dynamic_tab, text="Time: 0.0 s")
time_label.grid(row=2, column=0, columnspan=2, pady=5)

flow_label = tk.Label(dynamic_tab, text="Flow: 0.0 GPM")
flow_label.grid(row=3, column=0, columnspan=2, pady=5)

pos_label = tk.Label(dynamic_tab, text="Valve Pos: 0.0 %")
pos_label.grid(row=4, column=0, columnspan=2, pady=5)

u_label = tk.Label(dynamic_tab, text="Controller Out: 0.0 %")
u_label.grid(row=5, column=0, columnspan=2, pady=5)

model_d = None
running = False

def start_dynamic():
    global model_d, running
    try:
        model_d = DynamicModel()
        running = True
        update_dynamic()
    except ValueError:
        messagebox.showerror("Error", "Invalid input values")

def pause_dynamic():
    global running
    running = False

def reset_dynamic():
    global running
    running = False
    if model_d:
        model_d.reset()
    time_label.config(text="Time: 0.0 s")
    flow_label.config(text="Flow: 0.0 GPM")
    pos_label.config(text="Valve Pos: 0.0 %")
    u_label.config(text="Controller Out: 0.0 %")

def update_dynamic():
    if running:
        try:
            setpoint = float(setpoint_entry.get())
            pit = float(pit_d_entry.get())
            res = model_d.step(setpoint, pit)
            time_label.config(text=f"Time: {res['time_s']} s")
            flow_label.config(text=f"Flow: {res['flow_gpm']} GPM")
            pos_label.config(text=f"Valve Pos: {res['valve_position_pct']} %")
            u_label.config(text=f"Controller Out: {res['controller_output_pct']} %")
            dynamic_tab.after(int(model_d.dt * 1000), update_dynamic)
        except ValueError:
            messagebox.showerror("Error", "Invalid input values")
            pause_dynamic()

start_btn = tk.Button(dynamic_tab, text="Start", command=start_dynamic)
start_btn.grid(row=6, column=0, pady=10)

pause_btn = tk.Button(dynamic_tab, text="Pause", command=pause_dynamic)
pause_btn.grid(row=6, column=1, pady=10)

reset_btn = tk.Button(dynamic_tab, text="Reset", command=reset_dynamic)
reset_btn.grid(row=6, column=2, pady=10)

root.mainloop()
