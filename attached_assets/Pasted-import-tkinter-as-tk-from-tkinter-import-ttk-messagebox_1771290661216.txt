# Double-click editing
def edit_cell(event):
    item = inlet_tree.identify_row(event.y)
    col = inlet_tree.identify_column(event.x)
    if not item or col == "#1": return
    x, y, w, h = inlet_tree.bbox(item, col)
    val = inlet_tree.item(item)["values"][int(col[1:])-1]
    entry = Entry(inlet_frame, bg="#3a3a3a", fg="white")
    entry.place(x=x, y=y, width=w, height=h)
    entry.insert(0, val)
    entry.focus()

    def save(e=None):
        new_val = entry.get()
        vals = list(inlet_tree.item(item)["values"])
        vals[int(col[1:])-1] = new_val
        inlet_tree.item(item, values=vals)
        entry.destroy()

    entry.bind("<Return>", save)
    entry.bind("<FocusOut>", save)

inlet_tree.bind("<Double-1>", edit_cell)
