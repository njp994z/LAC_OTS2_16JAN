import tkinter as tk
from tkinter import messagebox

# Back-end function (copied for self-containment; in practice, import from back-end file)
def calculate_sulfuric_acid_parameters(sulfur_flow_lb_min, mol_so2_to_scrub, mol_so3_to_scrub, mol_o2_to_scrub, mol_n2_to_scrub):
    import math
    
    MW_S = 32.0
    MW_SO2 = 64.0
    MW_H2SO4 = 98.0
    MINUTES_PER_DAY = 1440
    LB_PER_SHORT_TON = 2000
    STEAM_GEN_RATE = 1.3
    POWER_FACTOR = 192.6
    SCRUBBER_SO2_REMOVAL = 0.97
    
    mol_s_min = sulfur_flow_lb_min / MW_S
    so2_conversion = (1 - mol_so2_to_scrub / mol_s_min) * 100
    mol_acid_min = mol_s_min - mol_so2_to_scrub - mol_so3_to_scrub
    lb_acid_day = mol_acid_min * MW_H2SO4 * MINUTES_PER_DAY
    plant_rate_stpd = lb_acid_day / LB_PER_SHORT_TON
    total_mol_to_scrub = mol_so2_to_scrub + mol_so3_to_scrub + mol_o2_to_scrub + mol_n2_to_scrub
    o2_consumed = 1.5 * mol_s_min - 0.5 * mol_so2_to_scrub
    o2_initial = mol_o2_to_scrub + o2_consumed
    mol_air = o2_initial / 0.21
    pass1_strength = (mol_s_min / mol_air) * 100
    emitted_mol_so2_min = (1 - SCRUBBER_SO2_REMOVAL) * mol_so2_to_scrub
    lb_so2_day = emitted_mol_so2_min * MW_SO2 * MINUTES_PER_DAY
    emissions_lb_stpd = lb_so2_day / plant_rate_stpd if plant_rate_stpd > 0 else 0
    scrubber_emissions_ppmv = (emitted_mol_so2_min / total_mol_to_scrub) * 1e6
    steam_gen_st_st = STEAM_GEN_RATE
    acid_st_per_hour = plant_rate_stpd / 24
    steam_st_per_hour = STEAM_GEN_RATE * acid_st_per_hour
    gross_power_kw = POWER_FACTOR * steam_st_per_hour
    gross_power_mw = gross_power_kw / 1000
    
    return {
        'Plant Rate (STPD)': round(plant_rate_stpd, 0),
        'Pass 1 Strength (% SO2)': round(pass1_strength, 1),
        'SO2 Conversion (%)': round(so2_conversion, 1),
        'Emissions (lb / STPD)': round(emissions_lb_stpd, 0),
        'Scrubber Emissions (ppmv)': round(scrubber_emissions_ppmv, 0),
        'Steam Gen. (ST/ST)': round(steam_gen_st_st, 2),
        'Gross Power Gen. (MW)': round(gross_power_mw, 3)
    }

# GUI setup
def create_gui():
    root = tk.Tk()
    root.title("Sulfuric Acid Plant Parameters Calculator")
    
    # Input labels and entries
    tk.Label(root, text="Sulfur Flow (lb/min):").grid(row=0, column=0, padx=10, pady=5)
    sulfur_entry = tk.Entry(root)
    sulfur_entry.grid(row=0, column=1)
    
    tk.Label(root, text="Mol SO2 to Scrubber (lb-mol/min):").grid(row=1, column=0, padx=10, pady=5)
    so2_entry = tk.Entry(root)
    so2_entry.grid(row=1, column=1)
    
    tk.Label(root, text="Mol SO3 to Scrubber (lb-mol/min):").grid(row=2, column=0, padx=10, pady=5)
    so3_entry = tk.Entry(root)
    so3_entry.grid(row=2, column=1)
    
    tk.Label(root, text="Mol O2 to Scrubber (lb-mol/min):").grid(row=3, column=0, padx=10, pady=5)
    o2_entry = tk.Entry(root)
    o2_entry.grid(row=3, column=1)
    
    tk.Label(root, text="Mol N2 to Scrubber (lb-mol/min):").grid(row=4, column=0, padx=10, pady=5)
    n2_entry = tk.Entry(root)
    n2_entry.grid(row=4, column=1)
    
    # Result display area
    result_text = tk.Text(root, height=10, width=50)
    result_text.grid(row=6, column=0, columnspan=2, padx=10, pady=10)
    
    def calculate():
        try:
            sulfur = float(sulfur_entry.get())
            so2 = float(so2_entry.get())
            so3 = float(so3_entry.get())
            o2 = float(o2_entry.get())
            n2 = float(n2_entry.get())
            
            params = calculate_sulfuric_acid_parameters(sulfur, so2, so3, o2, n2)
            
            result_text.delete(1.0, tk.END)
            for key, value in params.items():
                result_text.insert(tk.END, f"{key}: {value}\n")
        except ValueError:
            messagebox.showerror("Input Error", "Please enter valid numerical values.")
    
    # Calculate button
    calc_button = tk.Button(root, text="Calculate", command=calculate)
    calc_button.grid(row=5, column=0, columnspan=2, pady=10)
    
    root.mainloop()

# Run the GUI
if __name__ == "__main__":
    create_gui()
