"""
Main Compressor Performance Calculator - Fixed & Improved
Uses system curve + blower curve intersection to determine operating flow
"""

import math
import json
import sys
import numpy as np
from dataclasses import dataclass, asdict
from typing import Tuple, Optional

# ────────────────────────────────────────────────
# Constants
# ────────────────────────────────────────────────
R_AIR       = 53.35          # ft·lbf/(lbm·°R)
GAMMA       = 1.4
MW_AIR      = 28.97
INWC_TO_PSI = 0.03613
PSI_PER_ATM = 14.696

# Reference point (highest reliable point from datasheet - Case 6 clean)
Q_REF_ACFM   = 185802.0      # approx from pt6
N_REF_RPM    = 4505.0
DP_REF_INWC  = 575.4862 - 328.4862   # ~247 inwc (absolute - absolute)

# Rough system curve reference point (normal clean operation ~ Case 3/4)
Q_SYSTEM_REF_SCFM = 115000.0
DP_SYSTEM_REF_INWC_CLEAN = 199.0     # discharge - inlet gauge (from datasheet Case 3)

# Dirty multiplier (example - adjust based on your plant data)
DIRTY_FACTOR = {"clean": 1.0, "dirty": 1.45}

# Exponents
SYSTEM_EXPONENT   = 1.70
BLOWER_EXP_PRESS  = -1.68     # ≈ value from your original fit (negative = inverse)

# ────────────────────────────────────────────────
@dataclass
class CompressorInput:
    rpm_percent: float          # 0–100 % of maximum speed
    inlet_temp_F: float
    inlet_pressure_inwc_gauge: float   # inlet gauge pressure (negative = suction)
    barometric_atm: float
    plant_condition: str = "clean"     # "clean" or "dirty"

# ────────────────────────────────────────────────
@dataclass
class CompressorOutput:
    inlet_flow_acfm: float
    inlet_flow_am3hr: float
    outlet_temp_F: float
    outlet_temp_C: float
    outlet_pressure_inwc: float
    outlet_pressure_mmwg: float
    temp_rise_F: float
    temp_rise_C: float
    pressure_rise_inwc: float
    pressure_rise_mmwg: float
    standard_flow_scfm: float
    standard_flow_nm3hr: float
    outlet_flow_acfm: float
    outlet_flow_am3hr: float
    mass_flow_klbhr: float
    mass_flow_MThr: float
    isentropic_head_ftlblb: float
    isentropic_head_kJkg: float
    brake_power_hp: float
    brake_power_MW: float
    motor_power_hp: float
    motor_power_MW: float
    driver_speed_rpm: float

# ────────────────────────────────────────────────
# Unit conversion helpers
# ────────────────────────────────────────────────

def inwc_to_psia(inwc_g: float, baro_atm: float) -> float:
    return baro_atm * PSI_PER_ATM + inwc_g * INWC_TO_PSI

def fahrenheit_to_celsius(f: float) -> float:
    return (f - 32) * 5 / 9

def acfm_to_am3hr(acfm: float) -> float:
    return acfm * 0.0283168 * 60

def scfm_to_nm3hr(scfm: float) -> float:
    return scfm * 0.0283168 * 60 * (273.15 / 288.71)

def inwc_to_mmwg(inwc: float) -> float:
    return inwc * 25.4

# ────────────────────────────────────────────────
# Core calculation
# ────────────────────────────────────────────────

def find_operating_flow(
    rpm_percent: float,
    plant_condition: str = "clean"
) -> Tuple[float, float]:   # Q_acfm, ΔP_inwc (blower - system)
    N_ratio = rpm_percent / 100.0 * (N_REF_RPM / N_REF_RPM)  # usually just rpm_percent/100
    Q_guess = Q_REF_ACFM * N_ratio

    dirty_mult = DIRTY_FACTOR.get(plant_condition.lower(), 1.0)

    def blower_dp(Q: float) -> float:
        return DP_REF_INWC * (N_ratio ** 2) * (Q / Q_REF_ACFM) ** BLOWER_EXP_PRESS

    def system_dp(Q: float) -> float:
        return DP_SYSTEM_REF_INWC_CLEAN * dirty_mult * (Q / Q_SYSTEM_REF_SCFM) ** SYSTEM_EXPONENT

    # Simple fixed-point iteration (can be replaced with scipy.optimize later)
    Q = Q_guess
    for _ in range(40):
        dp_b = blower_dp(Q)
        dp_s = system_dp(Q)
        Q_new = Q * (dp_b / dp_s) ** (1 / (2 + BLOWER_EXP_PRESS - SYSTEM_EXPONENT))
        if abs(Q_new - Q) < 0.5:
            break
        Q = 0.65 * Q + 0.35 * Q_new
    else:
        print("Warning: iteration did not converge well", file=sys.stderr)

    dp_operating = system_dp(Q)   # = blower_dp(Q) at intersection
    return Q, dp_operating

# ────────────────────────────────────────────────
def calculate_compressor_performance(inp: CompressorInput) -> CompressorOutput:
    inlet_temp_R = inp.inlet_temp_F + 459.67
    inlet_psia   = inwc_to_psia(inp.inlet_pressure_inwc_gauge, inp.barometric_atm)

    baro_inwc_abs = inp.barometric_atm * PSI_PER_ATM / INWC_TO_PSI
    inlet_abs_inwc = inp.inlet_pressure_inwc_gauge + baro_inwc_abs

    # Find operating point
    Q_acfm, pressure_rise_inwc = find_operating_flow(
        inp.rpm_percent,
        inp.plant_condition
    )

    # Very approximate temperature rise (scale from reference point)
    temp_rise_F_approx = 123.0 * (Q_acfm / Q_REF_ACFM) ** 0.4   # rough scaling
    outlet_temp_F = inp.inlet_temp_F + temp_rise_F_approx

    outlet_temp_R = outlet_temp_F + 459.67

    outlet_psia = inlet_psia + pressure_rise_inwc * INWC_TO_PSI
    outlet_abs_inwc = inlet_abs_inwc + pressure_rise_inwc

    inlet_density = (inlet_psia * 144) / (R_AIR * inlet_temp_R)
    outlet_density = (outlet_psia * 144) / (R_AIR * outlet_temp_R)

    mass_flow_lbhr = Q_acfm * inlet_density * 60.0
    standard_flow_scfm = mass_flow_lbhr / (0.0808 * 60.0)      # rough 60°F, 14.7 psia

    outlet_flow_acfm = mass_flow_lbhr / (outlet_density * 60.0) if outlet_density > 0 else float("nan")

    # Very simplified head & power (adjust efficiency later)
    pressure_ratio = outlet_psia / inlet_psia if inlet_psia > 0 else 1.0
    isentropic_head = (GAMMA / (GAMMA - 1)) * R_AIR * inlet_temp_R * (pressure_ratio ** ((GAMMA-1)/GAMMA) - 1)

    efficiency = 0.78
    brake_power_hp = (mass_flow_lbhr / 60.0) * isentropic_head / (efficiency * 33000.0)
    motor_power_hp = brake_power_hp / 0.96

    gearbox_ratio = 4174.0 / 1654.0
    driver_speed = (inp.rpm_percent / 100.0 * N_REF_RPM) / gearbox_ratio

    return CompressorOutput(
        inlet_flow_acfm=Q_acfm,
        inlet_flow_am3hr=acfm_to_am3hr(Q_acfm),
        outlet_temp_F=outlet_temp_F,
        outlet_temp_C=fahrenheit_to_celsius(outlet_temp_F),
        outlet_pressure_inwc=outlet_abs_inwc - baro_inwc_abs,
        outlet_pressure_mmwg=inwc_to_mmwg(outlet_abs_inwc - baro_inwc_abs),
        temp_rise_F=temp_rise_F_approx,
        temp_rise_C=fahrenheit_to_celsius(temp_rise_F_approx),
        pressure_rise_inwc=pressure_rise_inwc,
        pressure_rise_mmwg=inwc_to_mmwg(pressure_rise_inwc),
        standard_flow_scfm=standard_flow_scfm,
        standard_flow_nm3hr=scfm_to_nm3hr(standard_flow_scfm),
        outlet_flow_acfm=outlet_flow_acfm,
        outlet_flow_am3hr=acfm_to_am3hr(outlet_flow_acfm),
        mass_flow_klbhr=mass_flow_lbhr / 1000,
        mass_flow_MThr=mass_flow_lbhr / 2204.62,
        isentropic_head_ftlblb=isentropic_head,
        isentropic_head_kJkg=isentropic_head * 0.00135582,
        brake_power_hp=brake_power_hp,
        brake_power_MW=brake_power_hp * 0.0007457,
        motor_power_hp=motor_power_hp,
        motor_power_MW=motor_power_hp * 0.0007457,
        driver_speed_rpm=driver_speed
    )

# ────────────────────────────────────────────────
# CLI / API entry point
# ────────────────────────────────────────────────
if __name__ == "__main__":
    try:
        data = json.load(sys.stdin)
        inp = CompressorInput(
            rpm_percent=float(data.get("rpm_percent", 88.0)),
            inlet_temp_F=float(data.get("temp", 150.0)),
            inlet_pressure_inwc_gauge=float(data.get("pressure", -12.0)),
            barometric_atm=float(data.get("barometricPressure", 0.85)),
            plant_condition=data.get("plant_condition", "clean")
        )
        result = calculate_compressor_performance(inp)
        output = asdict(result)
        for k, v in output.items():
            if isinstance(v, float) and not math.isfinite(v):
                output[k] = None
        print(json.dumps({"success": True, "results": output}, indent=2))
    except Exception as e:
        print(json.dumps({"success": False, "error": str(e)}))
        sys.exit(1)
