# sulfur_furnace_backend.py
import math

# Physical & engineering constants
MOL_S = 32.06               # lb/lb-mol sulfur
SCF_PER_LBMOL = 379.0       # standard cubic feet per lb-mol @ 60°F, 1 atm
O2_FRAC = 0.21
N2_FRAC = 0.79
HEAT_COMBUSTION = 4200.0    # BTU/lb S (approximate net heating value)
CP_GAS_AVG = 0.27           # BTU/lb·°F (approximate for SO₂ + air mixture)
T_AIR_INLET = 93.0          # °F (typical from data)

class SulfurFurnaceModel:
    def __init__(self):
        self.T_out_previous = 2000.0  # initial furnace outlet temperature guess (°F)
        self.tau = 600.0              # thermal time constant (seconds) for dynamic mode

    def calculate_static(
        self,
        air_scfm: float,
        sulfur_klb_hr: float
    ) -> dict:
        """Perform static (steady-state) material & heat balance."""
        m_sulfur_lbh = sulfur_klb_hr * 1000.0
        lb_mol_s_hr = m_sulfur_lbh / MOL_S

        # Volume flows (scfm) assuming complete conversion to SO₂
        scfm_so2 = lb_mol_s_hr * SCF_PER_LBMOL
        scfm_o2_consumed = scfm_so2
        scfm_o2_in = O2_FRAC * air_scfm
        scfm_o2_out = scfm_o2_in - scfm_o2_consumed
        scfm_n2 = N2_FRAC * air_scfm

        scfm_dry_total = scfm_so2 + scfm_o2_out + scfm_n2

        # Dry mole fractions
        pct_so2 = 100.0 * scfm_so2 / scfm_dry_total if scfm_dry_total > 0 else 0.0
        pct_o2  = 100.0 * scfm_o2_out / scfm_dry_total if scfm_dry_total > 0 else 0.0
        pct_n2  = 100.0 - pct_so2 - pct_o2

        # Approximate adiabatic furnace outlet temperature
        heat_release_btu_hr = m_sulfur_lbh * HEAT_COMBUSTION

        # Rough total gas mass flow (lb/hr) for heat capacity estimate
        mol_wt_air = 28.96
        mol_wt_so2 = 64.06
        mass_air = (air_scfm * mol_wt_air) / SCF_PER_LBMOL
        mass_so2  = (scfm_so2 * mol_wt_so2) / SCF_PER_LBMOL
        mass_gas_approx = mass_air + mass_so2

        delta_T = heat_release_btu_hr / (mass_gas_approx * CP_GAS_AVG) if mass_gas_approx > 0 else 0.0
        T_out = T_AIR_INLET + delta_T

        # Rough pressure estimate (linear scaling around example data)
        P_out_inwc = 180.0 + (scfm_dry_total - 115000.0) / 1000.0 * 10.0

        return {
            "mode": "Static",
            "air_scfm": air_scfm,
            "sulfur_klb_hr": sulfur_klb_hr,
            "scfm_so2": scfm_so2,
            "scfm_dry_total": scfm_dry_total,
            "pct_so2": pct_so2,
            "pct_o2": pct_o2,
            "pct_n2": pct_n2,
            "T_out_f": T_out,
            "P_out_inwc": P_out_inwc
        }

    def calculate_dynamic(
        self,
        air_scfm: float,
        sulfur_klb_hr: float,
        time_step_seconds: float
    ) -> dict:
        """Perform one dynamic time step with first-order temperature lag."""
        static_result = self.calculate_static(air_scfm, sulfur_klb_hr)
        T_ad = static_result["T_out_f"]

        # First-order lag: T(t) = T_prev + (T_ad - T_prev) * (1 - exp(-Δt/τ))
        exp_term = 1.0 - math.exp(-time_step_seconds / self.tau)
        T_new = self.T_out_previous + (T_ad - self.T_out_previous) * exp_term
        self.T_out_previous = T_new

        static_result["mode"] = "Dynamic"
        static_result["T_out_f"] = T_new
        return static_result

    def reset_dynamic_state(self):
        """Reset temperature state for a new dynamic simulation run."""
        self.T_out_previous = 2000.0
