import tkinter as tk
from tkinter import ttk, messagebox

def open_faceplate():
    faceplate = tk.Toplevel(root)
    faceplate.title("Faceplate - SIM SR PID 1")
    faceplate.geometry("380x620")
    faceplate.resizable(False, False)
    faceplate.configure(bg="#c0c0c0")
    faceplate.grab_set()          # makes it modal (real DeltaV feel)

    # === All the same faceplate code from before goes here ===
    # (I’m just copying the class contents into a function for simplicity)

    pv = tk.DoubleVar(value=170.0)
    sp = tk.DoubleVar(value=170.0)
    out = tk.DoubleVar(value=34.4)
    mode = tk.StringVar(value="AUTO")

    # Title
    tk.Label(faceplate, text="SIM SR PID 1", font=("Arial", 16, "bold"), bg="#c0c0c0").pack(pady=(20,0))
    tk.Label(faceplate, text="Self-regulating process,", font=("Arial", 10), bg="#c0c0c0").pack()

    # Top readouts
    top = tk.Frame(faceplate, bg="#c0c0c0")
    top.pack(pady=20)
    tk.Label(top, textvariable=out, font=("Arial", 20, "bold"), width=8, bg="black", fg="lime", relief="sunken").grid(row=0,column=0,padx=30)
    tk.Label(top, text="%", font=("Arial",12), bg="#c0c0c0").grid(row=1,column=0)
    sp_entry = tk.Entry(top, textvariable=sp, font=("Arial", 20, "bold"), width=8, justify="center", bg="black", fg="lime")
    sp_entry.grid(row=0,column=1,padx=30)
    tk.Label(top, text="°F", font=("Arial",12), bg="#c0c0c0").grid(row=1,column=1)

    # Bars (simplified but still nice)
    bars = tk.Frame(faceplate, bg="#c0c0c0")
    bars.pack(pady=20)
    out_canvas = tk.Canvas(bars, width=70, height=320, bg="white", highlightthickness=2, highlightbackground="black")
    out_canvas.pack(side="left", padx=40)
    out_fill = out_canvas.create_rectangle(8, 300, 62, 300, fill="#00ffff", outline="black")
    out_canvas.create_text(35, 15, text="100.0", font=("Arial",9))
    out_canvas.create_text(35, 310, text="0.0", font=("Arial",9))

    pv_canvas = tk.Canvas(bars, width=70, height=320, bg="white", highlightthickness=2, highlightbackground="black")
    pv_canvas.pack(side="right", padx=40)
    pv_fill = pv_canvas.create_rectangle(8, 20, 62, 300, fill="#ffff00", outline="black")
    pv_canvas.create_text(35, 15, text="200.0", font=("Arial",9))
    pv_canvas.create_text(35, 310, text="100.0", font=("Arial",9))

    # Mode buttons
    tk.Button(faceplate, text="AUTO", width=12, height=2, bg="#90ee90", font=("Arial",12,"bold"),
              command=lambda: messagebox.showinfo("Mode", "Switched to AUTO")).pack(pady=10)
    tk.Button(faceplate, text="MAN", width=12, height=2, bg="#ffcccc", font=("Arial",12,"bold"),
              command=lambda: messagebox.showinfo("Mode", "Switched to MAN")).pack(pady=5)

# === Main window (your future process graphic) ===
root = tk.Tk()
root.title("My DeltaV-Style Graphic")
root.geometry("1000x700")
root.configure(bg="#404040")

# Big button that looks like a tank or valve on a P&ID
big_button = tk.Button(root,
                       text="TIC-101\nClick for Faceplate",
                       font=("Arial", 20, "bold"),
                       width=15, height=6,
                       bg="#2080ff", fg="white",
                       relief="raised", bd=8,
                       command=open_faceplate)
big_button.pack(expand=True)

root.mainloop()
