from scipy.optimize import fsolve  # Requires scipy installed
import numpy as np

def solve_system(fcv_pos, tcv_pos, acid_flow_final, cooler_k, pump_a, pump_b, pump_c,

    - Valves are equal-percentage for Cv vs. position.

    - Cooler ΔP = cooler_k * flow_cooler**2
    - Pump discharge P = pump_a - pump_b * total_flow - pump_c * total_flow**2

        raise ValueError("Valve positions must be between 0 and 1.")

    # For better accuracy, use: rangeability = 50; Cv_tcv = cv_tcv_max * 50**(tcv_pos - 1)
    # But here, assume linear Cv vs. pos for simplicity
    cv_fcv = cv_fcv_max * fcv_pos
    cv_tcv = cv_tcv_max * tcv_pos

    # Per diagram, Acid Flow Final is input? But with pump curve, flow is determined by pump vs system.

        # Flow through bypass: standard valve equation flow = Cv * sqrt(dp / sg) (for liquids, in gpm? Wait units)
        # Caution: Units matter! Assume Cv in m³/h / bar^0.5, common in metric.
        flow_bypass = cv_tcv * np.sqrt(dp_common / fluid_sg)
        flow_cooler = total_flow - flow_bypass
        if flow_cooler < 0:
            return 1e6  # Invalid
        dp_cooler_calc = cooler_k * flow_cooler**2
        return dp_cooler_calc - dp_common

    # Solve for dp_common
    dp_common_guess = 1.0  # bar
    dp_common = fsolve(equations, dp_common_guess)[0]

    flow_bypass = cv_tcv * np.sqrt(dp_common / fluid_sg)
    flow_cooler = total_flow - flow_bypass

    # From diagram: Pump -> FCV -> Split (byp + cooler) -> P_final

    # Pump -> P_in (pump discharge?) -> FCV -> P_final? Wait, image shows Pump -> Cooler/TCV -> FCV ? Wait.

    # Pump discharge pressure (output1) pushes total_flow through the parallel (bypass + cooler), then mixed flow through FCV to P_final.

    # Assume P_final = 0 (atmospheric, or known backpressure).

    # Then, P_out_byp_cooler = P_final + dp_FCV
