def update_table1():
    # Re-read editable values and update table
    temp12 = hip_hot_temp.get()
    temp14 = cip_hot_temp.get()
    temp17a = cip_cold_temp.get()
    table1.item(table1.get_children()[8], values=("TEMPERATURE", "F", temp12, temp14, temp17a))

def run_static():
    update_table1()
    t3 = pass3_target.get()
    t4 = pass4_target.get()
    txt = f"Static Results:\nPass 3 Target Outlet: {t3} °F\nPass 4 Target Outlet: {t4} °F\n\nCalculated bypass flows and mixed temperatures would appear in Table 2."
    # Here: call StaticInterpassHX.calculate(float(t3), float(t4)) and update table2
    # For now placeholder update
    for i, item in enumerate(table2.get_children()):
        if i < 6:
            table2.item(item, values=table2.item(item)["values"][:3] + ("~806",))
        else:
            table2.item(item, values=table2.item(item)["values"][:3] + ("~779",))

def start_dynamic():
    global running
    running = True
    update_dynamic()

def pause_dynamic():
    global running
    running = False

def update_dynamic():
    if not running: return
    try:
        update_table1()
        ma = float(ma_entry.get())
        # Split-range example (simplified)
        if ma <= 12:
            pos_small = (ma - 4) / 8 * 100
            pos_large = 0
        else:
            pos_small = 100
            pos_large = (ma - 12) / 8 * 100

        # Placeholder updates for table2
        table2.item(table2.get_children()[6], values=("Pass 3 Mixed", "TEMPERATURE", "F", f"{float(hip_hot_temp.get()):.1f}"))
        table2.item(table2.get_children()[12], values=("Pass 4 Mixed", "TEMPERATURE", "F", f"{float(cip_hot_temp.get()):.1f}"))

        # Real dynamic update would come from DynamicInterpassHX.step(ma)
    except:
        pass
    root.after(1000, update_dynamic)

ttk.Button(btn_frame, text="Calculate Static", command=run_static).pack(side="left", padx=15)

ttk.Button(btn_frame, text="Pause", command=pause_dynamic).pack(side="left", padx=15)

running = False

# Mode toggle
def toggle_mode(*args):
    m = mode_var.get()
    static_frame.pack() if m == "Static" else static_frame.pack_forget()
    dynamic_frame.pack() if m == "Dynamic" else dynamic_frame.pack_forget()

mode_var.trace("w", toggle_mode)
toggle_mode()
