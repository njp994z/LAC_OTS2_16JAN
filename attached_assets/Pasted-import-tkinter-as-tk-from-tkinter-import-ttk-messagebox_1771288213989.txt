# ── Calculate function ───────────────────────────────────────────
def run_static_calculation():
    try:
        t12 = float(entries["Stream #12 HIP Hot Inlet Temp"].get())
        t14 = float(entries["Stream #14 CIP Hot Inlet Temp"].get())
        t17a = float(entries["Stream #17A CIP Cold Feed Temp"].get())
        target3 = float(pass3_target.get())
        target4 = float(pass4_target.get())

        # Update inlet table
        update_inlet_table()

        # Placeholder calculation logic (replace with real backend)
        hip_out_temp = target3 - 10      # example
        cip_out_temp = target4 - 8
        hip_bypass = 104164 * 0.35       # example 35%
        cip_bypass = 103736 * 0.42
        hip_hx_flow = 104164 - hip_bypass
        cip_hx_flow = 103736 - cip_bypass

        hip_pos = 65.0
        cip_36_pos = 100.0 if hip_pos > 50 else 0
        cip_78_pos = max(0, hip_pos - 50) * 2

        # Update output table
        values_to_set = [
            ("#13 HIP Hot Outlet",          "TEMPERATURE",      "F",      f"{hip_out_temp:.1f}"),
            ("#18A HIP Cold Feed",          "TEMPERATURE",      "F",      f"{t17a:.1f}"),
            ("#18B HIP Cold Inlet",         "TEMPERATURE",      "F",      f"{t17a:.1f}"),
            ("#18C HIP BYP",                "TOTAL FLOW",       "scfm",   f"{hip_bypass:,.0f}"),
            ("#18D HIP BYP Mix",            "TEMPERATURE",      "F",      f"{target3:.1f}"),
            ("#18E HIP Cold Outlet",        "TEMPERATURE",      "F",      f"{t17a + 120:.1f}"),
            ("#19 HIP Cold Mixed",          "TEMPERATURE",      "F",      f"{target3:.1f}"),
            ("#15 CIP Hot Outlet",          "TEMPERATURE",      "F",      f"{cip_out_temp:.1f}"),
            ("#17B CIP Cold Inlet",         "TEMPERATURE",      "F",      f"{t17a:.1f}"),
            ("#17C CIP BYP V_in",           "TOTAL FLOW",       "scfm",   f"{cip_bypass:,.0f}"),
            ("#17D CIP BYP V_out",          "TOTAL FLOW",       "scfm",   f"{cip_bypass:,.0f}"),
            ("#17E CIP Cold Outlet",        "TEMPERATURE",      "F",      f"{t17a + 140:.1f}"),
            ("#18 CIP Cold Out",            "TEMPERATURE",      "F",      f"{target4:.1f}"),
            ("HIP 48\" Valve Position",     "% open",           "",       f"{hip_pos:.1f}"),
            ("CIP 36\" Valve Position",     "% open",           "",       f"{cip_36_pos:.1f}"),
            ("CIP 78\" Valve Position",     "% open",           "",       f"{cip_78_pos:.1f}"),
            ("HIP HX / Bypass ratio",       "fraction",         "",       f"{hip_hx_flow/104164:.3f}"),
            ("CIP HX / Bypass ratio",       "fraction",         "",       f"{cip_hx_flow/103736:.3f}"),
        ]

        for i, (stream, param, unit, val) in enumerate(values_to_set):
            item = output_tree.get_children()[i]
            output_tree.item(item, values=(stream, param, unit, val))

        messagebox.showinfo("Calculation Complete", "Output streams updated (placeholder values)")

    except ValueError as ve:
        messagebox.showerror("Input Error", f"Invalid number:\n{ve}")
    except Exception as e:
        messagebox.showerror("Error", str(e))

# ── Mode toggle ──────────────────────────────────────────────────
def toggle_mode(*args):
    mode = mode_var.get()
    if mode == "Static":
        target_frame.pack(fill="x", padx=20, pady=6)
        dynamic_frame.pack_forget()
    else:
        target_frame.pack_forget()
        dynamic_frame.pack(fill="x", padx=20, pady=6)

mode_var.trace("w", toggle_mode)
toggle_mode()

# ── Live update inlet table on temp change ──────────────────────
def on_change(*args):
    update_inlet_table()

for e in entries.values():
    e.bind("<KeyRelease>", on_change)
