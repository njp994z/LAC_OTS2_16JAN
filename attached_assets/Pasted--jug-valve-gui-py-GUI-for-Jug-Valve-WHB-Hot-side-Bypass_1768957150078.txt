# jug_valve_gui.py
# GUI for Jug Valve (WHB Hot-side Bypass) Simulator
# Matches provided screenshots exactly

import tkinter as tk
from tkinter import ttk, messagebox
from jug_valve_calc import calculate_jug_valve_flows

root = tk.Tk()
root.title("Jug Valve, WHB Hot-side, Positioner")
root.geometry("1100x820")
root.configure(bg="#1e1e2e")

# ── Title ────────────────────────────────────────────────────────────────────
title_label = tk.Label(root, text="Jug Valve, WHB Hot-side, Positioner", 
                       font=("Arial", 16, "bold"), bg="#1e1e2e", fg="#ffffff")
title_label.pack(pady=10)

# ── System Parameters ────────────────────────────────────────────────────────
params_frame = ttk.LabelFrame(root, text="System Parameters", padding=10)
params_frame.pack(fill="x", padx=15, pady=5)

params_data = [
    ("Jug Valve Diameter (Inches)",           "36.0"),
    ("Furnace Outlet Diameter (ft)",          "9.0"),
    ("WHB HT Area (ft²)",                     "9800"),
    ("Barometric P (psia)",                   "14.3"),
    ("Weather ZIP Code",                      "89801"),
    ("WHB Outlet Duct Dia. (ft)",             "7.5"),
    ("Jug Valve Cv_max",                      "12500"),
    ("WHB Uo (ft²·°F·hr / BTU)",              "16.0"),
    ("WHB Steam Pressure (psig)",             "915.0"),
]

param_entries = {}
row = 0
for label, default in params_data:
    ttk.Label(params_frame, text=label).grid(row=row, column=0, padx=10, pady=6, sticky="e")
    
    if "ZIP Code" in label:
        entry = ttk.Entry(params_frame, width=12)
        entry.insert(0, default)
        entry.grid(row=row, column=1, padx=10, pady=6, sticky="w")
        param_entries[label] = entry
    else:
        ttk.Label(params_frame, text=default, width=12, relief="sunken", anchor="w").grid(
            row=row, column=1, padx=10, pady=6, sticky="w")
    
    row += 1

# ── System Inputs ────────────────────────────────────────────────────────────
inputs_frame = ttk.LabelFrame(root, text="System Inputs", padding=10)
inputs_frame.pack(fill="x", padx=15, pady=5)

input_labels = ["Jug Valve Opening Percentage", "Positioner Valve Opening"]
input_defaults = ["10", "100"]

input_entries = {}
for i, (lbl, val) in enumerate(zip(input_labels, input_defaults)):
    ttk.Label(inputs_frame, text=lbl).grid(row=0, column=i*2, padx=15, pady=8, sticky="e")
    entry = ttk.Entry(inputs_frame, width=10)
    entry.insert(0, val)
    entry.grid(row=0, column=i*2+1, padx=10, pady=8)
    input_entries[lbl] = entry

# ── Simulation Outputs Table ─────────────────────────────────────────────────
outputs_frame = ttk.LabelFrame(root, text="Simulation Outputs - Static Mode", padding=10)
outputs_frame.pack(fill="both", expand=True, padx=15, pady=5)

headers = [
    "", "Units",
    "Furnace Outlet\n1540-TI-42220ABC\nGF1",
    "Waste Heat\nBoiler In\nGB0",
    "Jug Valve Inlet\nGJV0",
    "Waste Heat\nBoiler Out\nGB1",
    "Positioner\nOutlet\nGPV1",
    "Gas Pass\n1 Inlet\nGP10"
]

streams = ["GF1", "GB0", "GJV0", "GB1", "GPV1", "GP10"]

for col, hdr in enumerate(headers):
    ttk.Label(outputs_frame, text=hdr, font=("Arial", 10, "bold")).grid(
        row=0, column=col, padx=6, pady=6, sticky="nsew")

row_items = [
    ("SO2",    "scfm"),
    ("SO3",    "scfm"),
    ("O2",     "scfm"),
    ("N2",     "scfm"),
    ("H2O",    "scfm"),
    ("H2SO4",  "scfm"),
    ("TOTAL",  "scfm"),
    ("PRESSURE", "in. wc."),
    ("TEMPERATURE", "F"),
]

output_labels = {}
for r, (param, unit) in enumerate(row_items, start=1):
    ttk.Label(outputs_frame, text=param).grid(row=r, column=1, padx=5, pady=4, sticky="e")
    ttk.Label(outputs_frame, text=unit).grid(row=r, column=2, padx=5, pady=4, sticky="w")
    
    for c, stream in enumerate(streams, start=3):
        lbl = ttk.Label(outputs_frame, text="---", width=12, relief="sunken", anchor="center")
        lbl.grid(row=r, column=c, padx=4, pady=3, sticky="nsew")
        output_labels[f"{param}_{stream}"] = lbl

# ── Calculation Trigger ──────────────────────────────────────────────────────
def run_calculation():
    try:
        jug_open_pct = float(input_entries["Jug Valve Opening Percentage"].get())
        pos_open_pct = float(input_entries["Positioner Valve Opening"].get())

        # Get fixed parameters (could be made editable if desired)
        cv_max = 12500
        u_value = 16.0
        whb_area = 9800
        baro_psia = 14.3

        results = calculate_jug_valve_flows(
            furnace_outlet_scfm_dry=109697,   # from HMB stream 9 dry total
            furnace_outlet_so2=12401,
            furnace_outlet_so3=227,
            furnace_outlet_o2=10261,
            furnace_outlet_n2=86808,
            furnace_outlet_temp_f=2080,
            furnace_outlet_press_inwc=196,
            jug_open_pct=jug_open_pct,
            positioner_open_pct=pos_open_pct,
            cv_max=cv_max,
            u_value=u_value,
            whb_area=whb_area,
            baro_psia=baro_psia
        )

        # Update GUI labels
        for key, value in results.items():
            if key in output_labels:
                output_labels[key].config(text=f"{value:.0f}" if "scfm" in key else f"{value:.1f}")

    except ValueError as e:
        messagebox.showerror("Input Error", "Please enter valid numbers for percentages.")
    except Exception as e:
        messagebox.showerror("Calculation Error", str(e))

# ── Buttons ──────────────────────────────────────────────────────────────────
btn_frame = ttk.Frame(root)
btn_frame.pack(pady=15)

ttk.Button(btn_frame, text="Calculate", command=run_calculation, width=15).pack(side="left", padx=20)
ttk.Button(btn_frame, text="Reset", command=lambda: None, width=15, state="disabled").pack(side="left", padx=20)

root.mainloop()
