import math
import time
import numpy as np
from scipy.interpolate import interp1d
from dataclasses import dataclass

CONFIG = {
    'pipe_dia_in': 4.0,
    'line_length_ft': 80.0,
    'deltaP_nozzle_psi': 150.0,
    'furnace_static_psi': 7.0,
    'friction_factor': 0.018,
    'K_minor_losses': 7.5,
    'SG': 1.79,
    'Cv_max': 548.0,
    'barometric_psig': 0.0,
    'temperature_f': 275.0
}

class PID:
    def __init__(self, Kp=1.2, Ki=0.15, Kd=0.0, setpoint=81.35):
        self.Kp = Kp
        self.Ki = Ki
        self.Kd = Kd
        self.setpoint = setpoint
        self.integral = 0.0
        self.previous_error = 0.0

    def update(self, measured_value, dt):
        error = self.setpoint - measured_value
        self.integral += error * dt
        derivative = (error - self.previous_error) / dt if dt > 0 else 0
        self.previous_error = error
        output = self.Kp * error + self.Ki * self.integral + self.Kd * derivative
        # Scale to 4-20 mA (assume raw output 0-100 -> 4-20 mA)
        mA = 4.0 + (min(max(output, 0.0), 100.0) / 100.0) * 16.0
        return max(4.0, min(20.0, mA))

@dataclass
class DynamicState:
    time_s: float = 0.0
    received_mA: float = 4.0
    positioner_pct: float = 0.0
    flow_gpm: float = 0.0

class DynamicSulfurSprayHydraulics:
    def __init__(self, valve_tau_s=8.0, flow_tau_s=4.0, dt=0.5, simulate_pid=False, pid_setpoint=81.35):
        self.dt = dt
        self.tau_valve = valve_tau_s
        self.tau_flow = flow_tau_s
        self.simulate_pid = simulate_pid
        self.pid = PID(Kp=1.2, Ki=0.15, Kd=0.0, setpoint=pid_setpoint) if simulate_pid else None
        self.state = DynamicState()
        self.start_time = time.time()

        self.psi_per_ft = CONFIG['SG'] * 0.433
        self.pipe_dia_ft = CONFIG['pipe_dia_in'] / 12.0
        self.pipe_area_ft2 = math.pi * (self.pipe_dia_ft / 2) ** 2
        self.pump_coeffs = [-1.06594794e-05, -7.73601399e-04, -4.91452991e-02, 296.104895]
        self.g = 32.174

        # Valve interpolation
        self.cv_points = np.array([4.549, 12.169, 25.039])
        self.pos_points = np.array([16.27, 23.4, 30.6])
        self.pos_to_cv = interp1d(self.pos_points, self.cv_points, kind='linear', fill_value="extrapolate")

    def mA_to_position(self, mA: float) -> float:
        return max(0.0, min(100.0, (mA - 4.0) / 16.0 * 100.0))

    def step(self, received_mA: float = None, pit_level_ft: float = 7.0) -> dict:
        self.state.time_s = time.time() - self.start_time

        if self.simulate_pid:
            # Approximate PID behavior
            received_mA = self.pid.update(self.state.flow_gpm, self.dt)

        self.state.received_mA = received_mA or self.state.received_mA

        # Positioner lag
        target_pos = self.mA_to_position(self.state.received_mA)
        self.state.positioner_pct += (target_pos - self.state.positioner_pct) * (1 - math.exp(-self.dt / self.tau_valve))
        self.state.positioner_pct = max(0.0, min(100.0, self.state.positioner_pct))

        # Cv from position (interpolated equal %)
        Cv = self.pos_to_cv(self.state.positioner_pct) if self.state.positioner_pct <= 30.6 else CONFIG['Cv_max'] * (self.state.positioner_pct / 100.0)

        # Calculate flow from Cv
        Q_guess = self.state.flow_gpm or 81.35
        for _ in range(20):
            a, b, c, d = self.pump_coeffs
            pump_head_ft = a * Q_guess**3 + b * Q_guess**2 + c * Q_guess + d

            Q_cfs = Q_guess / 448.831
            v_fps = Q_cfs / self.pipe_area_ft2
            h_f = (CONFIG['friction_factor'] * CONFIG['line_length_ft'] / self.pipe_dia_ft +
                   CONFIG['K_minor_losses']) * (v_fps**2) / (2 * self.g)

            suction_psig = pit_level_ft * self.psi_per_ft + CONFIG['barometric_psig']
            pump_discharge_psig = suction_psig + pump_head_ft * self.psi_per_ft - h_f * self.psi_per_ft

            discharge_line_loss_psi = 2.0
            valve_inlet_psig = pump_discharge_psig - discharge_line_loss_psi

            nozzle_inlet_psig = CONFIG['furnace_static_psi'] + CONFIG['deltaP_nozzle_psi']
            dp_valve_psi = max(valve_inlet_psig - nozzle_inlet_psig, 0.01)

            Q_new = Cv * math.sqrt(dp_valve_psi / CONFIG['SG'])

            if abs(Q_new - Q_guess) < 0.1:
                break
            Q_guess = 0.5 * Q_guess + 0.5 * Q_new

        # Flow lag
        self.state.flow_gpm += (Q_new - self.state.flow_gpm) * (1 - math.exp(-self.dt / self.tau_flow))

        # Populate material balance table
        flow_gpm = self.state.flow_gpm
        mass_lb_per_hr = flow_gpm * CONFIG['SG'] * 500.3
        mass_klb_per_hr = mass_lb_per_hr / 1000

        pump_discharge_psia = pump_discharge_psig + 14.7
        valve_inlet_psia = valve_inlet_psig + 14.7
        nozzle_inlet_psia = nozzle_inlet_psig + 14.7

        table = {
            "1540-PI-2600_psia": round(pump_discharge_psia, 2),
            "1540-FIC-2602_gpm": round(flow_gpm, 2),
            "1540-PI-2604_psia": round(nozzle_inlet_psia, 2),

            "x_H2SO4": 0.00,
            "x_H2O": 0.00,
            "x_Sulfur": 1.00,
            "m_Total_klb_hr": round(mass_klb_per_hr, 2),

            "Pressure_psia_pump": round(pump_discharge_psia, 2),
            "Pressure_psia_valve": round(valve_inlet_psia, 2),
            "Pressure_psia_nozzle": round(nozzle_inlet_psia, 2),

            "Temp_F": CONFIG['temperature_f'],
            "Flow_gpm_pump": round(flow_gpm, 2),
            "Flow_gpm_valve": round(flow_gpm, 2),
            "Flow_gpm_nozzle": round(flow_gpm, 2),

            "pump_head_ft": round(pump_head_ft, 2),
            "valve_position_pct": round(self.state.positioner_pct, 2),
            "Cv_used": round(Cv, 2),
            "dp_valve_psi": round(dp_valve_psi, 2),
            "pid_output_mA": round(self.state.received_mA, 2),
            "timestamp": datetime.now().isoformat()
        }

        return table

    def reset(self):
        self.state = DynamicState()
        self.start_time = time.time()
        if self.pid:
            self.pid.integral = 0.0
            self.pid.previous_error = 0.0
